# B站 AI 自动评论机器人（大肥鱼）—— 运行依赖
# 环境：Python 3.10+（本机 3.12.10 已验证）
# 安装：python -m venv .venv && .venv\Scripts\activate && pip install -r requirements.txt

# ---------------------------------------------------------------------------
# 1. B站接口封装（推荐流 / 视频信息 / 发评论 / 消息中心）
#    钉死小版本：这套库 17.x 重构过网络层（bilibili_api.clients），
#    不同小版本的函数签名会变，升级前请重新对照文档。
bilibili-api-python==17.4.2

# ---------------------------------------------------------------------------
# 2. bilibili-api-python 没有把 HTTP 客户端写进依赖，需要自己装一个。
#    它会在 curl_cffi / aiohttp / httpx 里自动挑一个，都没有就直接抛 ArgsException。
#    这里固定用 httpx（openai 也依赖它，不会多装东西）。
httpx>=0.27,<1.0

# ---------------------------------------------------------------------------
# 3. 扫码登录要把二维码存成 PNG（bilibili_api 的 Picture.to_file 依赖 Pillow）
pillow>=9.1

# ---------------------------------------------------------------------------
# 4. 调 DeepSeek（OpenAI 兼容接口，base_url = https://api.deepseek.com/v1）
#    钉在 1.x：DeepSeek 官方示例走的就是这条线，chat.completions.create 稳定。
openai>=1.55,<2.0

# ---------------------------------------------------------------------------
# 4. 关闭窗口后留在系统托盘（仅 Windows；macOS 上不启用托盘）
pystray>=0.19; sys_platform == "win32"

# ---------------------------------------------------------------------------
# 5. 读取 .env（源码运行时可用，打包后设置都在界面里）
python-dotenv>=1.0,<2.0
